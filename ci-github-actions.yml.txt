# GitHub Actions CI (Three-Gate v7) — preserved as a plain file.
# To activate: move to .github/workflows/ci.yml and push with a PAT that has
# BOTH `repo` and `workflow` scopes (this push used a repo-only token).
#
name: ZoningFight CI (Three-Gate v7)

on:
  push:
    branches: [master, main]
  pull_request:

jobs:
  gates:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - name: Install
        run: npm ci || npm install
      - name: Gate 2 — static (B1–B12)
        run: node scripts/gate2.mjs
      - name: Gate 1 — sims (full, incl. host URLs)
        run: node scripts/verify.mjs
      - name: Build
        run: npm run build
